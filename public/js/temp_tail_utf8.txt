        if (templateId === 'thpt_50') {

            p3Container?.classList.remove('d-none');
        } else {
            p2Container?.classList.add('d-none');
            p3Container?.classList.add('d-none');
        }
    }

    // Enable next button
    document.getElementById('btn-go-step2').disabled = false;

    // Using toast.success assuming it exists
    if (typeof toast !== 'undefined' && toast.success) {
        toast.success(`Đã chọn: ${templates[templateId]?.name}`);
    }
}
